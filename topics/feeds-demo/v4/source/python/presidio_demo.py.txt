"""Local field anonymization called by .NET through Python.NET.

.NET owns entire-feed parsing, binary serialization, validation and delivery.
No JSON feed processing, tokenization, identity mapping or remote calls are used.
"""
import importlib.util
from presidio_analyzer import AnalyzerEngine, RecognizerResult
from presidio_anonymizer import AnonymizerEngine
from presidio_anonymizer.entities import OperatorConfig

# Require a preinstalled model; never download models while processing feeds.
if importlib.util.find_spec("en_core_web_lg") is None:
    raise RuntimeError("Install the bundled en_core_web_lg model before starting the listener.")

analyzer = AnalyzerEngine()
anonymizer = AnonymizerEngine()

KNOWN_FIELDS = {
    **dict.fromkeys(["name", "fullname", "passengername", "firstname", "lastname", "surname", "givenname"], "PERSON"),
    **dict.fromkeys(["email", "emailaddress"], "EMAIL_ADDRESS"),
    **dict.fromkeys(["phone", "phonenumber", "mobile"], "PHONE_NUMBER"),
    **dict.fromkeys(["creditcard", "cardnumber"], "CREDIT_CARD"),
    **dict.fromkeys(["passport", "passportnumber"], "PASSPORT_NUMBER"),
    **dict.fromkeys(["sourcepassengerid", "passengerid"], "PASSENGER_ID"),
    "address": "ADDRESS", "dateofbirth": "DATE_OF_BIRTH", "dob": "DATE_OF_BIRTH",
    "ticketnumber": "TICKET_NUMBER",
}
PNR_FIELDS = {"pnr", "linkedpnr", "oldpnr", "newpnr"}
TECHNICAL_FIELDS = {
    "tenant", "messageid", "feedtype", "flight", "flightnumber", "status", "seat",
    "linkreason", "eventtype", "timestamp", "departure", "arrival", "origin", "destination",
}
OPERATORS = {
    **{entity: OperatorConfig("replace", {"new_value": "[" + entity + "]"})
       for entity in set(KNOWN_FIELDS.values())},
    "PERSON": OperatorConfig("replace", {"new_value": "[PASSENGER_NAME]"}),
    "DEFAULT": OperatorConfig("replace"),
}


def anonymize_feed_field(value, field):
    """Anonymize one decoded text value while preserving operational/PNR fields."""
    if not isinstance(value, str) or not isinstance(field, str):
        raise ValueError("Decoded text and a field name are required")
    key = field.casefold()
    if key in PNR_FIELDS or key in TECHNICAL_FIELDS or not value:
        return value
    if key in KNOWN_FIELDS:
        results = [RecognizerResult(KNOWN_FIELDS[key], 0, len(value), 1.0)]
    else:
        results = analyzer.analyze(text=value, language="en", context=[field])
    return anonymizer.anonymize(text=value, analyzer_results=results, operators=OPERATORS).text
